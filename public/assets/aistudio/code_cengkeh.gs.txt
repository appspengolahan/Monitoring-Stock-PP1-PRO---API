var SHEET_TARGET = [
  "MANADO GRADE A (2024)",
  "MANADO GRADE A (2023)",
  "GAGANG PELURU A 2018",
  "MANADO KODE B"
];

// Tab yang dikecualikan dari pemrosesan
var TAB_DIKECUALIKAN = [
  "1",
  "KARTU STOK SPAREPART",
  "VALIDASI CROSSCHECK"
];

var KOLOM_TANGGAL    = 2; // Kolom B: HARI DAN TANGGAL
var KOLOM_PEMICU     = 3; // Kolom C: JENIS MUTASI
var BARIS_HEADER     = 7; // Baris 7: Header Tabel
var BARIS_RINGKASAN  = 8; // Baris 8: Subheader / Ringkasan Tanggal
var BARIS_MULAI_DATA = 9; // Baris 9: Mulai data mutasi transaksi
var BARIS_PROTEKSI   = 8; // Baris 1-8 diproteksi (tidak boleh di-hide)
var BATAS_HARI_HIDE  = 6; // Sembunyikan baris mutasi lebih dari 6 hari

var NAMA_HARI = ["Minggu", "Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu"];
var NAMA_BULAN = [
  "Januari", "Februari", "Maret", "April", "Mei", "Juni",
  "Juli", "Agustus", "September", "Oktober", "November", "Desember"
];
var PETA_BULAN = {
  "Januari": 0, "Februari": 1, "Maret": 2, "April": 3, "Mei": 4, "Juni": 5,
  "Juli": 6, "Agustus": 7, "September": 8, "Oktober": 9, "November": 10, "Desember": 11
};

// =====================================================================
// 1. SIMPLE TRIGGER: ON EDIT (PENGISIAN TANGGAL OTOMATIS)
// =====================================================================
/**
 * Berjalan otomatis saat ada perubahan sel oleh operator:
 * Jika operator mengisi JENIS MUTASI (Kolom C) dan HARI DAN TANGGAL (Kolom B) kosong,
 * script secara otomatis mengisi Kolom B dengan tanggal terakhir yang ada di atasnya.
 */
function onEdit(e) {
  if (!e || !e.range) return;
  var sheet = e.range.getSheet();
  var sheetName = sheet.getName();

  // Validasi apakah sheet termasuk kartu stok Cengkeh
  if (!isSheetTarget(sheetName)) return;

  var baris = e.range.getRow();
  var kolom = e.range.getColumn();

  // Hanya proses jika yang diedit adalah Kolom C dan berada pada baris data (>= 9)
  if (baris < BARIS_MULAI_DATA) return;
  if (kolom !== KOLOM_PEMICU) return;

  var nilaiC = e.value;
  if (nilaiC === undefined || nilaiC === null || String(nilaiC).trim() === "") return;

  // Cek apakah Kolom B pada baris ini sudah ada isinya
  var selTanggal = sheet.getRange(baris, KOLOM_TANGGAL);
  var nilaiB = selTanggal.getValue();
  if (nilaiB !== "" && nilaiB !== null && nilaiB !== undefined) return;

  // Cari tanggal terakhir dari baris tepat di atasnya
  var tanggalAcuan = cariTanggalTerakhirDiAtas(sheet, baris);

  // Jika di atasnya belum ada tanggal, fallback ke hari ini
  if (!tanggalAcuan) {
    tanggalAcuan = formatTanggalSekarang();
  }

  // Tulis tanggal ke kolom B
  selTanggal.setValue(tanggalAcuan);
  SpreadsheetApp.flush();
}

/**
 * Mencari tanggal terakhir yang terisi pada kolom B mulai dari baris - 1
 * mundur ke atas hingga baris 8 (baris subheader/tanggal awal).
 */
function cariTanggalTerakhirDiAtas(sheet, barisTarget) {
  if (barisTarget <= BARIS_RINGKASAN) return null;

  var jumlahBarisDiAtas = barisTarget - BARIS_RINGKASAN;
  var rentangB = sheet.getRange(BARIS_RINGKASAN, KOLOM_TANGGAL, jumlahBarisDiAtas, 1).getValues();

  for (var i = rentangB.length - 1; i >= 0; i--) {
    var val = rentangB[i][0];
    if (val !== "" && val !== null && val !== undefined) {
      if (val instanceof Date) {
        return NAMA_HARI[val.getDay()] + ", " + padZero(val.getDate()) + " " + NAMA_BULAN[val.getMonth()] + " " + val.getFullYear();
      }
      return String(val).trim();
    }
  }
  return null;
}

function formatTanggalSekarang() {
  var d = new Date();
  return NAMA_HARI[d.getDay()] + ", " + padZero(d.getDate()) + " " + NAMA_BULAN[d.getMonth()] + " " + d.getFullYear();
}

function padZero(num) {
  return num < 10 ? "0" + num : String(num);
}

function isSheetTarget(namaSheet) {
  if (TAB_DIKECUALIKAN.indexOf(namaSheet) !== -1) return false;
  return SHEET_TARGET.indexOf(namaSheet) !== -1;
}

// =====================================================================
// 2. FUNGSI MASSAL: PERBAIKI SEMUA TANGGAL KOSONG PADA SELURUH TAB
// =====================================================================
/**
 * Menyisir seluruh tab Cengkeh dari baris 9 s/d baris terakhir.
 * Jika kolom C berisi transaksi namun kolom B kosong, kolom B akan
 * otomatis diisi dengan tanggal transaksi terakhir yang ada di atasnya.
 */
function perbaikiSemuaTanggalKosong() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var totalDiperbaikiSemua = 0;
  var rincianLaporan = [];

  SHEET_TARGET.forEach(function (namaSheet) {
    try {
      var sheet = ss.getSheetByName(namaSheet);
      if (!sheet) return;

      var lastRow = sheet.getLastRow();
      if (lastRow < BARIS_MULAI_DATA) return;

      var jumlahBaris = lastRow - BARIS_RINGKASAN + 1;
      var rentangB = sheet.getRange(BARIS_RINGKASAN, KOLOM_TANGGAL, jumlahBaris, 1).getValues();
      var rentangC = sheet.getRange(BARIS_RINGKASAN, KOLOM_PEMICU, jumlahBaris, 1).getValues();

      var tanggalBerjalan = null;
      var diperbaikiDiSheetIni = 0;

      for (var i = 0; i < jumlahBaris; i++) {
        var barisAktual = BARIS_RINGKASAN + i;
        var valB = rentangB[i][0];
        var valC = rentangC[i][0];

        // Jika baris ini memiliki tanggal, simpan sebagai tanggal acuan berjalan
        if (valB !== "" && valB !== null && valB !== undefined) {
          tanggalBerjalan = (valB instanceof Date)
            ? (NAMA_HARI[valB.getDay()] + ", " + padZero(valB.getDate()) + " " + NAMA_BULAN[valB.getMonth()] + " " + valB.getFullYear())
            : String(valB).trim();
        } else if (barisAktual >= BARIS_MULAI_DATA) {
          // Kolom B kosong pada baris transaksi
          var adaTransaksi = valC !== "" && valC !== null && String(valC).trim() !== "";
          if (adaTransaksi && tanggalBerjalan) {
            sheet.getRange(barisAktual, KOLOM_TANGGAL).setValue(tanggalBerjalan);
            diperbaikiDiSheetIni++;
          }
        }
      }

      if (diperbaikiDiSheetIni > 0) {
        totalDiperbaikiSemua += diperbaikiDiSheetIni;
        rincianLaporan.push("• " + namaSheet + ": " + diperbaikiDiSheetIni + " baris diperbaiki");
      }
    } catch (err) {
      Logger.log("Gagal memperbaiki sheet '" + namaSheet + "': " + err.message);
    }
  });

  var info = "Perbaikan Tanggal Selesai!\nTotal " + totalDiperbaikiSemua + " baris tanggal berhasil diisi.";
  if (rincianLaporan.length > 0) {
    info += "\n\nRincian Tab:\n" + rincianLaporan.join("\n");
  } else {
    info += "\n(Semua baris transaksi sudah memiliki tanggal yang valid)";
  }

  Logger.log(info);
  try {
    SpreadsheetApp.getUi().alert("Hasil Perbaikan Tanggal Cengkeh", info, SpreadsheetApp.getUi().ButtonSet.OK);
  } catch (e) {}
}

// =====================================================================
// 3. JARING PENGAMAN: PERBAIKAN TANGGAL OTOMATIS BERKALA (TIMER)
// =====================================================================
function perbaikanTanggalOtomatis() {
  var props = PropertiesService.getScriptProperties();

  SHEET_TARGET.forEach(function (namaSheet) {
    try {
      var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(namaSheet);
      if (!sheet) return;

      var lastRow = sheet.getLastRow();
      if (lastRow < BARIS_MULAI_DATA) return;

      var keyProp = "batasBaris_Cengkeh_" + namaSheet;
      var batasSebelumnya = parseInt(props.getProperty(keyProp), 10);

      if (isNaN(batasSebelumnya)) {
        props.setProperty(keyProp, String(lastRow));
        return;
      }

      if (lastRow <= batasSebelumnya) return;

      var mulai = batasSebelumnya + 1;
      var jumlahBaris = lastRow - batasSebelumnya;

      var nilaiB = sheet.getRange(mulai, KOLOM_TANGGAL, jumlahBaris, 1).getValues();
      var nilaiC = sheet.getRange(mulai, KOLOM_PEMICU, jumlahBaris, 1).getValues();

      var jumlahDiperbaiki = 0;

      for (var i = 0; i < nilaiC.length; i++) {
        var baris = mulai + i;
        var adaC = nilaiC[i][0] !== "" && nilaiC[i][0] !== null;
        var kosongB = nilaiB[i][0] === "" || nilaiB[i][0] === null;

        if (adaC && kosongB) {
          var tglAcuan = cariTanggalTerakhirDiAtas(sheet, baris) || formatTanggalSekarang();
          sheet.getRange(baris, KOLOM_TANGGAL).setValue(tglAcuan);
          jumlahDiperbaiki++;
        }
      }

      props.setProperty(keyProp, String(lastRow));

      if (jumlahDiperbaiki > 0) {
        Logger.log("Sheet '" + namaSheet + "': " + jumlahDiperbaiki + " baris baru diperbaiki.");
      }
    } catch (err) {
      Logger.log("Gagal memproses sheet '" + namaSheet + "': " + err.message);
    }
  });
}

function inisialisasiBatasBaris() {
  var props = PropertiesService.getScriptProperties();
  SHEET_TARGET.forEach(function (namaSheet) {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(namaSheet);
    if (!sheet) return;
    var lastRow = sheet.getLastRow();
    props.setProperty("batasBaris_Cengkeh_" + namaSheet, String(lastRow));
  });
  Logger.log("Inisialisasi baseline baris Cengkeh selesai.");
}

// 4. AUTO HIDE: SEMBUNYIKAN BARIS LEBIH DARI 6 HARI
// =====================================================================
function hideBarisLama() {
  var sheetGagal = [];

  SHEET_TARGET.forEach(function (namaSheet) {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName(namaSheet);
      if (!sheet) return;

      var lastRow = sheet.getLastRow();
      var maxRows = sheet.getMaxRows();
      if (maxRows <= BARIS_PROTEKSI) return;

      // Buka terlebih dahulu semua baris di bawah header
      sheet.showRows(BARIS_PROTEKSI + 1, maxRows - BARIS_PROTEKSI);

      if (lastRow < BARIS_MULAI_DATA) return;

      var jumlahBaris = lastRow - BARIS_MULAI_DATA + 1;
      var nilaiB = sheet.getRange(BARIS_MULAI_DATA, KOLOM_TANGGAL, jumlahBaris, 1).getValues();
      var nilaiC = sheet.getRange(BARIS_MULAI_DATA, KOLOM_PEMICU, jumlahBaris, 1).getValues();

      var sekarang = new Date();
      var tanggalBerjalan = null;
      var barisUntukHide = [];

      for (var i = 0; i < jumlahBaris; i++) {
        var baris = BARIS_MULAI_DATA + i;
        var tanggalDariBaris = ambilTanggalDariNilai(nilaiB[i][0]);
        if (tanggalDariBaris) tanggalBerjalan = tanggalDariBaris;

        var adaData = nilaiC[i][0] !== "" && nilaiC[i][0] !== null;
        if (!adaData || !tanggalBerjalan) continue;

        var selisihHari = Math.floor((sekarang - tanggalBerjalan) / (1000 * 60 * 60 * 24));
        if (selisihHari > BATAS_HARI_HIDE) {
          barisUntukHide.push(baris);
        }
      }

      var totalHide = hideBarisBatch(sheet, barisUntukHide);
      Logger.log("Sheet '" + namaSheet + "': " + totalHide + " baris disembunyikan.");
    } catch (err) {
      sheetGagal.push(namaSheet);
      Logger.log("Gagal hide baris pada sheet '" + namaSheet + "': " + err.message);
    }
  });

  if (sheetGagal.length > 0) {
    Logger.log("Sheet yang gagal di-hide: " + sheetGagal.join(", "));
  }
}

function hideBarisBatch(sheet, daftarBaris) {
  if (!daftarBaris || daftarBaris.length === 0) return 0;

  var totalHide = 0;
  var awal = daftarBaris[0];
  var sebelumnya = daftarBaris[0];

  for (var i = 1; i <= daftarBaris.length; i++) {
    var barisSekarang = daftarBaris[i];
    if (barisSekarang === sebelumnya + 1) {
      sebelumnya = barisSekarang;
      continue;
    }
    sheet.hideRows(awal, sebelumnya - awal + 1);
    totalHide += (sebelumnya - awal + 1);

    awal = barisSekarang;
    sebelumnya = barisSekarang;
  }

  return totalHide;
}

// 5. HELPER PEMBACAAN TANGGAL
// =====================================================================
function ambilTanggalDariNilai(nilai) {
  if (nilai === "" || nilai === null || nilai === undefined) return null;
  if (nilai instanceof Date) {
    return new Date(nilai.getFullYear(), nilai.getMonth(), nilai.getDate());
  }
  if (typeof nilai === "string" && nilai.trim() !== "") {
    return parseTanggalIndo(nilai);
  }
  return null;
}

function parseTanggalIndo(teks) {
  var bagian = teks.split(",");
  if (bagian.length < 2) return null;
  var sisa = bagian[1].trim().split(" ");
  if (sisa.length < 3) return null;
  var tanggal = parseInt(sisa[0], 10);
  var bulan = PETA_BULAN[sisa[1]];
  var tahun = parseInt(sisa[2], 10);
  if (isNaN(tanggal) || bulan === undefined || isNaN(tahun)) return null;
  return new Date(tahun, bulan, tanggal);
}

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("Manajemen Mutasi Cengkeh")
    .addItem("Perbaiki Semua Tanggal Kosong (Isi dari Baris Atas)", "perbaikiSemuaTanggalKosong")
    .addItem("Tampilkan Semua Baris (Unhide)", "tampilkanSemuaBaris")
    .addItem("Sembunyikan Baris Lama (> 6 Hari)", "hideBarisLama")
    .addToUi();
}

function tampilkanSemuaBaris() {
  SHEET_TARGET.forEach(function (namaSheet) {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(namaSheet);
    if (!sheet) return;
    var maxRows = sheet.getMaxRows();
    if (maxRows > BARIS_PROTEKSI) {
      sheet.showRows(BARIS_PROTEKSI + 1, maxRows - BARIS_PROTEKSI);
    }
  });
  SpreadsheetApp.getUi().alert("Semua baris (di bawah baris proteksi 1-8) telah ditampilkan kembali.");
}
